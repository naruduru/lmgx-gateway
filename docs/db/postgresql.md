# PostgreSQL DDL

This project optionally writes health, failover, and instance status to PostgreSQL.
Use this schema if you enable DB logging.

Health check table stores the latest status by `(SRC_IP, TARGET_IP)` via upsert (no history accumulation).

```sql
CREATE TABLE IF NOT EXISTS GW_TARGET_HEALTH (
  SRC_IP VARCHAR(64) NOT NULL,
  TARGET_IP VARCHAR(64) NOT NULL,
  IS_UP CHAR(1) NOT NULL,
  UPDATED_AT TIMESTAMP NOT NULL DEFAULT NOW(),
  PRIMARY KEY (SRC_IP, TARGET_IP)
);
CREATE INDEX IF NOT EXISTS IX_GW_TH_01 ON GW_TARGET_HEALTH (UPDATED_AT);

CREATE TABLE IF NOT EXISTS GW_FAILOVER_EVENT (
  ID BIGSERIAL PRIMARY KEY,
  CREATED_AT TIMESTAMP NOT NULL DEFAULT NOW(),
  SERVER_GROUP VARCHAR(2) NOT NULL,
  FROM_TARGET VARCHAR(10),
  TO_TARGET VARCHAR(10) NOT NULL,
  FROM_URL VARCHAR(300),
  TO_URL VARCHAR(300),
  EVENT_KIND VARCHAR(10) NOT NULL,
  TRIGGER_REASON VARCHAR(200),
  FAIL_COUNT BIGINT,
  RECOVER_STABLE BIGINT
);
CREATE INDEX IF NOT EXISTS IX_GW_FE_01 ON GW_FAILOVER_EVENT (CREATED_AT);
CREATE INDEX IF NOT EXISTS IX_GW_FE_02 ON GW_FAILOVER_EVENT (SERVER_GROUP, CREATED_AT);

CREATE TABLE IF NOT EXISTS GW_INSTANCE_STATUS (
  INSTANCE_ID VARCHAR(50) PRIMARY KEY,
  GROUP_ID VARCHAR(2),
  SERVER_PORT INTEGER,
  ACTIVE_URL VARCHAR(300),
  ACTIVE_TARGET VARCHAR(10),
  CHAT_OPEN BOOLEAN,
  EMAIL_OPEN BOOLEAN,
  READY BOOLEAN,
  UPDATED_AT TIMESTAMP NOT NULL DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS IX_GW_IS_01 ON GW_INSTANCE_STATUS (UPDATED_AT);
```
